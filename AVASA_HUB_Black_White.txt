local function corner(obj, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 10)
    c.Parent = obj
    return c
end

local function outline(obj, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color or theme.stroke
    s.Thickness = thickness or 1
    s.Transparency = 0.15
    s.Parent = obj
    return s
end

local function makeLabel(parent, text, size, font, color)
    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextSize = size or 14
    label.Font = font or Enum.Font.Gotham
    label.TextColor3 = color or theme.text
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.TextWrapped = true
    label.Parent = parent
    return label
end

    label.Position = UDim2.new(0, 11, 0, 5)

    desc.Position = UDim2.new(0, 11, 0, 25)
    desc.Size = UDim2.new(1, -22, 0, 15)

    button.Activated:Connect(function()
        if callback then callback() end
    end)
    button.MouseEnter:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.12), {BackgroundColor3 = Color3.fromRGB(55, 55, 62)}):Play()
    end)
    button.MouseLeave:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.12), {BackgroundColor3 = theme.panel2}):Play()
    end)

    clearContent()
    for tabName, button in pairs(tabButtons) do
        button.BackgroundColor3 = (tabName == name) and theme.accent or theme.panel
        button.TextColor3 = theme.text
    end

    if name == "Overview" then
        addSection("WELCOME")
        addInfo("Fresh light-blue interface template. Add your own permitted features to the callbacks.")
        addSection("QUICK ACCESS")
        addAction("Interface status", "Confirms that the UI is responding", function()
            addInfo("Avasa UI is active.")
        end)
    elseif name == "Character" then
        addSection("Player")
        addInfo("Add safe settings for your own experience here.")
        addAction("Example action", "Placeholder — no gameplay effect", function()
            addInfo("This is a placeholder action.")
        end)
    elseif name == "Display" then
        addSection("Visuals")
        addInfo("Visual options for an experience you own can be placed here.")
        addAction("Example visual option", "Placeholder — connect your own implementation", function()
            addInfo("No visual setting is connected yet.")
        end)
    elseif name == "Tools" then
        addSection("LOCAL UTILITIES")
        addInfo("These buttons only control your own interface; they do not affect other players.")
        addAction("Refresh player count", "Show how many players are in this server", function()
            local count = #Players:GetPlayers()
            addInfo("Players in server: " .. tostring(count))
        end)
        addAction("My account info", "Display your local player name and user ID", function()
            addInfo("Username: " .. player.Name .. "  |  UserId: " .. tostring(player.UserId))
        end)
        addAction("Center window", "Move the menu back to the middle of the screen", function()
            main.AnchorPoint = Vector2.new(0.5, 0.5)
            main.Position = UDim2.fromScale(0.5, 0.5)
            addInfo("Window centered.")
        end)
        addAction("Toggle compact mode", "Switch between compact and normal menu size", function()
            if main:GetAttribute("CompactMode") then
                main:SetAttribute("CompactMode", false)
                main.Size = UDim2.new(0.9, 0, 0.7, 0)
                addInfo("Compact mode disabled.")
            else
                main:SetAttribute("CompactMode", true)
                main.Size = UDim2.new(0.82, 0, 0.58, 0)
                addInfo("Compact mode enabled.")
            end
        end)
        addAction("Show interface status", "Check that the menu is responding", function()
            addInfo("Avasa UI is running.")
        end)
    elseif name == "Options" then
        addSection("Settings")
        addAction("Hide window", "The small AVASA button will reopen it", function()
            main.Visible = false
        end)
        addAction("Reset interface", "Rebuild the current page", function()
            showPage("Options")
        end)
        addInfo("Safe interface utilities only. Game-cheat and anti-cheat bypass features are not included.")
    end
    content.CanvasPosition = Vector2.zero

local windowTween
local function playWindowTween(targetSize, targetTransparency, onComplete)
    if windowTween then
        pcall(function() windowTween:Cancel() end)
    end
    windowTween = TweenService:Create(
        main,
        TweenInfo.new(0.18, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
        {Size = targetSize, BackgroundTransparency = targetTransparency}
    )
    if onComplete then
        windowTween.Completed:Once(onComplete)
    end
    windowTween:Play()
end

local normalSize = UDim2.new(0.9, 0, 0.7, 0)
close.Activated:Connect(function()
    playWindowTween(UDim2.new(0.82, 0, 0.62, 0), 1, function()
        main.Visible = false
        main.BackgroundTransparency = 0
        main.Size = normalSize
        reopen.Visible = true
    end)
end)

reopen.Activated:Connect(function()
    reopen.Visible = false
    main.Size = UDim2.new(0.82, 0, 0.62, 0)
    main.BackgroundTransparency = 1
    main.Visible = true
    playWindowTween(normalSize, 0)
end)

-- Drag the top bar; supports mouse and touch.
do
    local dragging = false
    local dragInput
    local dragStart
    local startPosition

    top.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    top.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input == dragInput or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            main.Position = UDim2.new(
                startPosition.X.Scale, startPosition.X.Offset + delta.X,
                startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
            )
        end
    end)
end
